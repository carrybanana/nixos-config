/etc/nixos
├── flake.nix          # 入口：定义系统、主机、模块、home-manager
├── flake.lock         # 锁定版本，保证可复现
├── home/carry/        # 用户级配置（home-manager）
│   ├── kitty          # 终端配置
│   ├── secrets        # 密钥/敏感配置
│   └── shell          # shell 环境
├── hosts/             # 多主机配置（台式机 + 笔记本）
│   ├── desktop
│   └── laptop
├── modules/           # 系统全局模块（核心！）
│   ├── fonts          # 字体
│   ├── input-method   # 输入法（fcitx5）
│   ├── services       # 桌面环境、显示管理器
│   │   ├── desktop-managers/gnome、hyprland、kde、niri
│   │   └── display-managers/sddm 等
│   ├── system/programs  # 系统级软件（firefox、nvim、vscode、steam…）
│   └── virtualisation   # 虚拟化（qemu-kvm）
├── overlays/          # 包覆盖、自定义编译选项
└── Wallpapers/        # 壁纸
